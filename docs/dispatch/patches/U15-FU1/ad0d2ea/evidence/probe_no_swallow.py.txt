"""U15-FU1 probe ③：本轮改动的异常边界不得吞 KeyboardInterrupt/SystemExit。"""
from __future__ import annotations

import sys
from pathlib import Path

ROOT = Path(sys.argv[1]).resolve()
sys.path.insert(0, str(ROOT / "src"))

from core import release_manifest as rm  # noqa: E402
from core import runtime_identity as ri  # noqa: E402


def _run(label: str, exc_type) -> None:
    def boom(*_args, **_kwargs):
        raise exc_type("probe")

    target = "_direct_yaml" if hasattr(rm, "_direct_yaml") else "_git_commit"
    original = getattr(rm, target)
    setattr(rm, target, boom)
    try:
        rm.build_release_manifest(ROOT, environment="demo", runtime_instance_id="rt-probe")
    except exc_type:
        print(f"   {label}: propagated through _identity_evidence (OK)")
    except BaseException as exc:  # noqa: BLE001
        print(f"   {label}: unexpected {type(exc).__name__}: {exc}")
    else:
        print(f"   {label}: SWALLOWED (FAIL)")
    finally:
        setattr(rm, target, original)


def main() -> int:
    _run("KeyboardInterrupt", KeyboardInterrupt)
    _run("SystemExit", SystemExit)
    # 归一化函数本身不得屏蔽 BaseException（无异常边界）
    try:
        ri.resolve_runtime_instance_id("unknown")
    except BaseException as exc:  # noqa: BLE001
        print(f"   resolve raised {type(exc).__name__}: {exc}")
    else:
        print("   resolve_runtime_instance_id('unknown') -> None branch OK")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

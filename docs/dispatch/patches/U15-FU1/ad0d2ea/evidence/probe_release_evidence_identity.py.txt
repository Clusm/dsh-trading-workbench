"""U15-FU1 probe ②：占位符修前/修后在 **U09 发布证据门禁** 的读数差异。"""
from __future__ import annotations

import sys
from pathlib import Path

ROOT = Path(sys.argv[1]).resolve()
sys.path.insert(0, str(ROOT / "src"))

from core import release_evidence as ev  # noqa: E402
from core.release_manifest import build_release_manifest  # noqa: E402


def main() -> int:
    for requested in (None, "unknown", " rt-real "):
        manifest = build_release_manifest(
            ROOT, environment="demo", runtime_instance_id=requested).to_dict()
        block = ev._identity_block(
            manifest, release_id=str(manifest.get("release_id") or ""),
            environment="demo", account_ref=str(manifest.get("account_ref") or ""))
        print(f"requested={requested!r}")
        print(f"   manifest.runtime_instance_id = {manifest.get('runtime_instance_id')!r}")
        print(f"   manifest.identity_blockers  = {list(manifest.get('identity_blockers') or [])}")
        print(f"   gate.state                  = {block.get('state')!r}")
        print(f"   gate.reason_codes           = {list(block.get('reason_codes') or [])}")
        print(f"   gate.runtime_instance_id    = {block.get('runtime_instance_id')!r}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

"""Q 独立探针 N2：service.py 启动(:314) / set_control(:1416) 两条路径。

自建 spy（不抄作者探针），额外构造"行为可区分窗口"候选：
  A 未收口 ENTRY 只落在 -wal（作者场景）   → 判两态行为是否相同
  B 账本干净                                → 对照，不得误伤
  C 未收口 **非 ENTRY**（EXIT）单已 checkpoint 落主文件
      门禁账本读 = UNKNOWN(所有 purpose) → 裸形态阻断
      服务注入   = 只数 ENTRY            → 注入形态放行
      → 若成立，即"接线形态"确实可产生可达行为差异

全程离线：只写系统临时目录里的合成状态根；不读真实 .env/data/密钥；不联网/下单。
用法: python probe_n2.py <tree_root>
"""
from __future__ import annotations

import json
import os
import shutil
import sqlite3
import sys
import tempfile
import time
from pathlib import Path

TREE = Path(sys.argv[1]).resolve()
sys.path.insert(0, str(TREE / "src"))
os.environ["EXECUTION_MODE"] = "paper"
os.environ.pop("MTA_PROFILE", None)
for key in ("MTA_STATE_ROOT", "MTA_DATA_ROOT"):
    os.environ.pop(key, None)

import yaml  # noqa: E402

from audit.database import AuditDatabase  # noqa: E402
from core.config_loader import Settings  # noqa: E402
from core.ops.recovery import RecoveryGate  # noqa: E402
from core.service import TradingService  # noqa: E402
from exchange.order_outbox import OrderOutbox  # noqa: E402

PASS = 0
FAIL: list[str] = []


def check(name: str, ok: bool, detail: str = "") -> None:
    global PASS
    if ok:
        PASS += 1
        print(f"  PASS  {name}")
    else:
        FAIL.append(name)
        print(f"  FAIL  {name}  {detail}")


def _settings(iso: Path) -> Settings:
    s = Settings(
        TREE,
        yaml.safe_load((TREE / "config/settings.yaml").read_text(encoding="utf-8")),
        yaml.safe_load((TREE / "config/agents.yaml").read_text(encoding="utf-8")),
        yaml.safe_load((TREE / "config/risk_learning.yaml").read_text(encoding="utf-8")),
    )
    s.project_root = iso
    return s


def build_root(base: Path, name: str, *, mode: str) -> Path:
    """mode: 'wal_entry' | 'clean' | 'unknown_exit'"""
    iso = base / name
    data = iso / "data"
    (data / "ops").mkdir(parents=True, exist_ok=True)
    db_path = data / "agent_trader.sqlite"
    db = AuditDatabase(db_path)
    db.environment = "paper"
    staged: Path | None = None
    if mode == "wal_entry":
        db.conn.execute("PRAGMA journal_mode=WAL")
        OrderOutbox(db, environment="paper").prepare(
            client_order_id="ENTRY-N2", symbol="BTCUSDT", side="BUY",
            position_side="LONG", order_type="MARKET", purpose="ENTRY",
            qty=0.01, price=None, reduce_only=False,
            endpoint="/fapi/v1/order", request={}, command_kind="STANDARD",
            qty_text="0.01")
        db.conn.commit()
        staged = base / f"{name}_staging"
        staged.mkdir(parents=True, exist_ok=True)
        for m in ("agent_trader.sqlite", "agent_trader.sqlite-wal"):
            shutil.copy2(data / m, staged / m)
    db.conn.commit()
    db.close()
    if staged is not None:
        for m in ("agent_trader.sqlite", "agent_trader.sqlite-wal"):
            shutil.copy2(staged / m, data / m)

    for n, payload in {
        "capital_state.json": {"mode": "paper"},
        "channel_state.json": {"date": "2026-09-20"},
        "control_state.json": {"state": "normal"},
        "account_risk_state.json": {"anchor_id": "paper:q"},
    }.items():
        (data / n).write_text(json.dumps(payload), encoding="utf-8")

    now_ms = int(time.time() * 1000)
    gate_file = data / "ops" / "recovery_gate.json"
    RecoveryGate.load(gate_file).require_reconcile(
        epoch_id="e-q", account_ref="", runtime_instance_id="")
    dec = RecoveryGate.load(gate_file).complete_reconcile(
        {"observed_at_ms": now_ms, "positions_verified": True,
         "open_orders_verified": True}, now_ms=now_ms)
    print(f"  [{name}] 门禁建立 => allowed={dec.allowed} codes={dec.reason_codes}")

    if mode == "unknown_exit":
        # 门禁 CLEARED 之后再出现一颗 UNKNOWN 的 EXIT 单（已 checkpoint 落主文件）
        con = sqlite3.connect(str(db_path))
        con.execute("PRAGMA journal_mode=WAL")
        con.execute(
            "INSERT INTO order_outbox (client_order_id,environment,business_key,"
            "symbol,side,position_side,order_type,purpose,qty,price,reduce_only,"
            "endpoint,status,request_json,attempt_count,created_at,updated_at) "
            "VALUES ('EXIT-N2','paper','EXIT:BTCUSDT:LONG','BTCUSDT','SELL','LONG',"
            "'MARKET','EXIT',0.01,NULL,1,'/fapi/v1/order','UNKNOWN','{}',1,"
            "'2026-09-21T00:00:00Z','2026-09-21T00:00:00Z')")
        con.commit()
        con.execute("PRAGMA wal_checkpoint(TRUNCATE)")
        con.close()
    print(f"  [{name}] 落盘文件 = {sorted(p.name for p in data.iterdir())}")
    return iso


def run_case(base: Path, name: str, *, mode: str) -> dict:
    print(f"\n--- case {name} (mode={mode}) ---")
    iso = build_root(base, name, mode=mode)
    calls: list[object] = []
    original = RecoveryGate.before_entry

    def spy(self, *a, **kw):  # noqa: ANN001
        calls.append(kw.get("facts"))
        return original(self, *a, **kw)

    RecoveryGate.before_entry = spy
    try:
        service = TradingService(_settings(iso))
    finally:
        RecoveryGate.before_entry = original
    startup_calls = len(calls)
    startup_bare = sum(1 for f in calls if f is None)
    print(f"  启动期 before_entry x{startup_calls}, 未注入 {startup_bare}, "
          f"注入的事实 {[f for f in calls if f is not None]}")

    res = {"startup_calls": startup_calls, "startup_bare": startup_bare,
           "control": service.control.state,
           "gate_allowed": service._entry_readiness_facts.get("recovery_gate_allowed"),
           "gate_codes": list(
               service._entry_readiness_facts.get("recovery_gate_reason_codes") or []),
           "same_ledger": (Path(service.runtime_paths.db_file)
                           == RecoveryGate.load(
                               service.runtime_paths.recovery_gate_file).ledger_file())}

    calls.clear()
    original = RecoveryGate.before_entry
    RecoveryGate.before_entry = spy
    try:
        sr = service.set_control("normal")
    finally:
        RecoveryGate.before_entry = original
    res["set_control"] = sr
    res["set_control_calls"] = len(calls)
    res["set_control_bare"] = sum(1 for f in calls if f is None)
    print(f"  set_control('normal') -> {sr}")
    print(f"  set_control 期 x{len(calls)}, 未注入 {res['set_control_bare']}, "
          f"注入的事实 {[f for f in calls if f is not None]}")
    try:
        service.close()
    except Exception as exc:  # noqa: BLE001
        print(f"  NOTE close(): {type(exc).__name__}")
    return res


def main() -> int:
    base = Path(tempfile.mkdtemp(prefix="q-n2-"))
    print(f"tree = {TREE}\ntmp  = {base}")
    try:
        a = run_case(base, "wal_entry", mode="wal_entry")
        print("\n[W] 接线形态")
        check("启动路径注入事实(0 次裸调用)", a["startup_bare"] == 0 and a["startup_calls"] >= 1,
              f"calls={a['startup_calls']} bare={a['startup_bare']}")
        check("set_control 路径注入事实", a["set_control_bare"] == 0 and a["set_control_calls"] >= 1,
              f"calls={a['set_control_calls']} bare={a['set_control_bare']}")
        check("服务账本 == 门禁账本(同一文件)", a["same_ledger"])
        print("\n[A] -wal 里未收口 ENTRY")
        check("A 启动期门禁阻断", a["gate_allowed"] is False,
              f"allowed={a['gate_allowed']} codes={a['gate_codes']} control={a['control']}")

        b = run_case(base, "clean", mode="clean")
        print("\n[B] 对照：干净账本")
        check("B 启动期门禁放行", b["gate_allowed"] is True,
              f"allowed={b['gate_allowed']} codes={b['gate_codes']}")
        check("B 控制状态保持 normal", b["control"] == "normal", b["control"])
        check("B set_control('normal') 成功", b["set_control"].get("ok") is True,
              str(b["set_control"]))

        c = run_case(base, "unknown_exit", mode="unknown_exit")
        print("\n[C] 未收口 **非 ENTRY**(EXIT, UNKNOWN) 单 —— 可区分窗口候选")
        print(f"  C 结果: gate_allowed={c['gate_allowed']} codes={c['gate_codes']} "
              f"control={c['control']} set_control={c['set_control']}")
        check("C 启动期门禁阻断(裸形态语义)", c["gate_allowed"] is False,
              f"allowed={c['gate_allowed']} codes={c['gate_codes']}")
        check("C 控制状态收紧为 close_only", c["control"] == "close_only", c["control"])
        check("C set_control('normal') 被拒", c["set_control"].get("ok") is False,
              str(c["set_control"]))
    finally:
        shutil.rmtree(base, ignore_errors=True)
    print(f"\nPROBE N2: {PASS} 通过 / {len(FAIL)} 失败")
    for n in FAIL:
        print(f"  FAILED: {n}")
    return 1 if FAIL else 0


if __name__ == "__main__":
    raise SystemExit(main())

"""V1b 映射覆盖率自检。

对每个 P1/P2 拆分候选模块：把该模块**源码里所有顶层公开定义**（AST 取 FunctionDef/
ClassDef/Assign/AnnAssign 的名字，过滤 `_` 前缀）与本文档中**反引号点名过的符号**求差。
差额非空 = 映射漏块（U11 R1 的 B8-B13 / F6 就是这样漏掉的）。
"""
import ast, re, sys
from pathlib import Path

DOC = Path("docs/重构接口映射.md")
CANDIDATES = [("core.decision_trace", "src/core/decision_trace.py"),
              ("core.strategy_funnel", "src/core/strategy_funnel.py")]

text = DOC.read_text(encoding="utf-8")
mentioned = set()
for tok in re.findall(r"`([^`\n]+)`", text):
    mentioned |= set(re.findall(r"[A-Za-z_][A-Za-z0-9_]*", tok))

def top_level_public(path):
    tree = ast.parse(Path(path).read_text(encoding="utf-8"))
    out = {}
    for node in tree.body:
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
            names = (node.name,)
        elif isinstance(node, ast.Assign):
            names = tuple(t.id for t in node.targets if isinstance(t, ast.Name))
        elif isinstance(node, ast.AnnAssign) and isinstance(node.target, ast.Name):
            names = (node.target.id,)
        else:
            names = ()
        for n in names:
            if not n.startswith("_"):
                out[n] = node.lineno
    return out

bad = 0
for mod, src in CANDIDATES:
    defs = top_level_public(src)
    missing = sorted(n for n in defs if n not in mentioned)
    pct = 100.0 * (len(defs) - len(missing)) / max(1, len(defs))
    print(f"{mod}: 顶层公开定义 {len(defs)}，文中点名 {len(defs)-len(missing)}，"
          f"覆盖率 {pct:.1f}%，漏项 {len(missing)}")
    if missing:
        bad += 1
        for n in missing:
            print(f"   MISSING: {n}  (首次定义于 {src}:{defs[n]})")
print("V1B " + ("OK 映射覆盖完整" if not bad else f"FAIL {bad} 个模块有漏项"))
sys.exit(1 if bad else 0)

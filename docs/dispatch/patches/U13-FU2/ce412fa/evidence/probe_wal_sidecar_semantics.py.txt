import shutil, sqlite3, sys, tempfile
from pathlib import Path
TREE = Path(sys.argv[1]).resolve()
sys.path.insert(0, str(TREE / "src"))
from core.ops.recovery import ledger_reconcile_facts
tmp = Path(tempfile.mkdtemp(prefix="wal-exp-"))
a = tmp / "a"; a.mkdir()
src = a / "agent_trader.sqlite"
c = sqlite3.connect(src)
c.execute("PRAGMA journal_mode=WAL")
c.execute("CREATE TABLE order_outbox(status TEXT, purpose TEXT, environment TEXT)")
c.execute("INSERT INTO order_outbox VALUES('SUBMITTING','ENTRY','demo')")
c.commit()
print("A files:", sorted(p.name for p in a.iterdir()))
b = tmp / "b"; b.mkdir()
for name in ("agent_trader.sqlite", "agent_trader.sqlite-wal"):
    shutil.copy2(a / name, b / name)
print("B files:", sorted(p.name for p in b.iterdir()))
c.close()
dst = b / "agent_trader.sqlite"
print("immutable=1 read :", ledger_reconcile_facts(dst))
n = sqlite3.connect(dst)
try:
    print("normal open read :", n.execute("SELECT COUNT(*) FROM order_outbox").fetchone())
finally:
    n.close()
print("B files after normal open:", sorted(p.name for p in b.iterdir()))
shutil.rmtree(tmp, ignore_errors=True)

"""U13-FU2 探针 N3：文档口径 vs 代码行为（``assert_environment_bound``）。

用法：``python probe_n3_doc_vs_code.py <tree_root>``

对照两边：

- **代码行为**：``RuntimePaths.resolve(<代码根>, profile="demo")``（未声明状态根）
  的 ``validate("demo")`` 返回什么、``assert_environment_bound("demo")`` 抛不抛。
- **文档口径**：U13-FU1 补丁包 README 有没有还写着"demo 只是被检出并报告、
  不阻止启动"。

只读被检树，不写任何文件、不联网、不启动服务。
"""
from __future__ import annotations

import json
import sys
from pathlib import Path

TREE = Path(sys.argv[1]).resolve()
sys.path.insert(0, str(TREE / "src"))

from core.runtime_paths import RuntimePaths, RuntimePathsError  # noqa: E402

README = TREE / "docs/dispatch/patches/U13-FU1/bc112fe/README.md"

PASS = 0
FAIL: list[str] = []


def check(name: str, ok: bool, detail: str = "") -> None:
    global PASS
    if ok:
        PASS += 1
        print(f"  PASS  {name}")
    else:
        FAIL.append(name)
        print(f"  FAIL  {name}  {detail}")


def _find_u13fu1_entry(node: object) -> dict | None:
    """在任务清单里定位 U13-FU1 那条登记（按 patch_package 指向）。"""
    if isinstance(node, dict):
        if "tradeoffs" in node and "U13-FU1" in str(node.get("patch_package") or ""):
            return node
        for value in node.values():
            found = _find_u13fu1_entry(value)
            if found is not None:
                return found
    elif isinstance(node, list):
        for value in node:
            found = _find_u13fu1_entry(value)
            if found is not None:
                return found
    return None


def main() -> int:
    import tempfile

    print(f"probe N3 tree = {TREE}")
    with tempfile.TemporaryDirectory(prefix="u13fu2-n3-") as tmp:
        code_root = Path(tmp) / "code_root"
        code_root.mkdir(parents=True, exist_ok=True)
        paths = RuntimePaths.resolve(code_root, profile="demo")
        reasons = paths.validate("demo")
        print(f"  validate('demo') = {reasons}")
        raised = ""
        try:
            paths.assert_environment_bound("demo")
        except RuntimePathsError as exc:
            raised = str(exc)
        print(f"  assert_environment_bound('demo') -> raised={bool(raised)}")
        if raised:
            print(f"    {raised}")
        check("assert_environment_bound('demo') 对未声明状态根**会抛**",
              bool(raised) and "STATE_ROOT_MISSING" in raised, raised or "no raise")

    text = README.read_text(encoding="utf-8") if README.is_file() else ""
    print(f"  README({README.name}) 提到 assert_environment_bound 会抛: "
          f"{'assert_environment_bound' in text}")
    check("README 明确写了 assert_environment_bound 对 demo **会抛**",
          "assert_environment_bound" in text and "会抛" in text,
          "README 未登记真实行为")

    # 旧措辞的原始出处：U13-FU1 在任务清单里登记的 tradeoff（只看那条取舍，
    # 不扫全文件——评审记录里引用旧措辞来描述缺陷是正当的）。
    registrations = sorted(TREE.glob("docs/*任务清单*.json"))
    print(f"  登记文件 = {[p.name for p in registrations]}")
    stale: list[str] = []
    for path in registrations:
        entry = _find_u13fu1_entry(json.loads(path.read_text(encoding="utf-8")))
        if entry is None:
            stale.append(f"{path.name}:未找到 U13-FU1 条目")
            continue
        for item in entry.get("tradeoffs") or []:
            if "只是被检出并报告" in item:
                stale.append(f"{path.name}: {item[:60]}...")
            if "assert_environment_bound" in item and "会抛" in item:
                print(f"  更正文案: {item[:120]}...")
    print(f"  U13-FU1 tradeoff 仍称 demo『被检出并报告』: {stale or '(无)'}")
    check("任务清单里 U13-FU1 的 tradeoff 不再声称 demo 只被检出/仅报告",
          not stale, f"stale={stale}")
    print(f"\nPROBE N3: {PASS} 通过 / {len(FAIL)} 失败")
    for name in FAIL:
        print(f"  FAILED: {name}")
    return 1 if FAIL else 0


if __name__ == "__main__":
    raise SystemExit(main())

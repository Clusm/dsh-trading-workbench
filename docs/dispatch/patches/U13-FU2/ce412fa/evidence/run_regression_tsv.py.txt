"""逐脚本跑官方离线回归清单（与官方入口 ``run_offline_regression.py`` 同环境），
把每个脚本的退出码写成 TSV：``<script>\t<rc>``。

用法：``python run_regression_tsv.py <tree_root> <out_tsv>``

脚本清单**从官方入口模块里读**（不另抄一份，避免漂移）；某脚本失败不中断，
以便逐脚本对比 pristine vs patched 的失败集合增量。
"""
from __future__ import annotations

import importlib.util
import os
import subprocess
import sys
from pathlib import Path

TREE = Path(sys.argv[1]).resolve()
OUT = Path(sys.argv[2]).resolve()

spec = importlib.util.spec_from_file_location(
    "official_regression", TREE / "scripts/run_offline_regression.py")
module = importlib.util.module_from_spec(spec)
spec.loader.exec_module(module)

env = os.environ.copy()
env["EXECUTION_MODE"] = "paper"
env.pop("MTA_PROFILE", None)
env.pop("MTA_STATE_ROOT", None)
env.pop("MTA_DATA_ROOT", None)
env["MTA_NO_PROXY"] = "1"
env["PYTHONUTF8"] = "1"
env["PYTHONIOENCODING"] = "utf-8"
for name in tuple(env):
    upper = name.upper()
    if ("BINANCE" in upper or "SENSENOVA" in upper) and (
            "KEY" in upper or "SECRET" in upper or "TOKEN" in upper):
        env.pop(name, None)

rows: list[str] = []
for script in module.SCRIPTS:
    proc = subprocess.run([sys.executable, script], cwd=TREE, env=env,
                          capture_output=True, text=True,
                          encoding="utf-8", errors="replace")
    rows.append(f"{script}\t{proc.returncode}")
    print(f"{script}\t{proc.returncode}", flush=True)
    if proc.returncode != 0:
        tail = (proc.stdout or "")[-600:] + (proc.stderr or "")[-600:]
        print(f"  --- tail({script}) ---\n{tail}\n  --- end ---", flush=True)
OUT.write_text("\n".join(rows) + "\n", encoding="utf-8")
print(f"WROTE {OUT} scripts={len(rows)}")

"""U15-FU3 两态探测：哨兵运行代在两个消费入口的行为（A/B + 反向证据）。

用法::

    python probe.py <tree_root>
"""
from __future__ import annotations

import json
import sys
import time
from pathlib import Path

ROOT = Path(sys.argv[1]).resolve()
sys.path.insert(0, str(ROOT / "src"))

from core.entry_readiness import (  # noqa: E402
    authorization_matches_context,
    evaluate_entry_readiness,
)
from core.trading_scope import TradingScope  # noqa: E402

SENTINEL = "unknown"
REAL = "rt-2026-09-22-a1b2c3"
LOOKALIKES = ("unknownverse", "known", "n/a-x", "UNKNOWN-X")


def facts(scope: TradingScope, **overrides: object) -> dict[str, object]:
    base: dict[str, object] = {
        "service_accepting": True,
        "control_state": "normal",
        "equity_available": True,
        "account_risk_allow_new": True,
        "startup_safety_ready": True,
        "order_submission_ready": True,
        "market_cooldown_active": False,
        "data_budget_available": True,
        "rate_coordination_available": True,
        "order_mirror_ok": True,
        "protection_mirror_ok": True,
        "unknown_orders": 0,
        "submitting_orders": 0,
        "scope_hash": scope.scope_hash,
        "persistence_available": True,
    }
    base.update(overrides)
    return base


class Decision:
    def __init__(self, *, expires_at: float, environment: str,
                 scope_hash: str, runtime_instance_id: str) -> None:
        self.pipeline_authorization_expires_at = expires_at
        self.pipeline_environment = environment
        self.pipeline_scope_hash = scope_hash
        self.pipeline_runtime_instance_id = runtime_instance_id


def emit(tag: str, value: object) -> None:
    print(f"{tag} {json.dumps(value, ensure_ascii=False, sort_keys=True)}")


def main() -> None:
    scope = TradingScope.from_symbols(["BTCUSDT", "ETHUSDT"])
    now = time.time()

    # --- A. evaluate_entry_readiness：哨兵 vs 真值 vs 像哨兵的真值 ---------
    for label, value in (("sentinel", SENTINEL), ("real", REAL),
                         *[(f"lookalike:{item}", item) for item in LOOKALIKES]):
        snap = evaluate_entry_readiness(
            facts(scope), now=now, scope=scope,
            runtime_instance_id=value, environment="demo")
        emit(f"[A:{label}]",
             {"allowed": snap.new_entry_allowed,
              "reasons": list(snap.blocking_reasons),
              "runtime_instance_id": snap.runtime_instance_id})

    # --- A2. 哨兵经 facts 通道（未传显式参数）-----------------------------
    snap = evaluate_entry_readiness(
        facts(scope, runtime_instance_id=SENTINEL), now=now, scope=scope,
        environment="demo")
    emit("[A:facts_channel]", {"allowed": snap.new_entry_allowed,
                               "reasons": list(snap.blocking_reasons),
                               "runtime_instance_id": snap.runtime_instance_id})

    # --- A3. 三态事实语义（应与修前逐字相同）-----------------------------
    tri = {}
    for key, values in (("order_mirror_ok", (True, False, None)),
                        ("protection_mirror_ok", (True, False, None)),
                        ("service_accepting", (True, False, None))):
        for value in values:
            snap = evaluate_entry_readiness(
                facts(scope, **{key: value}), now=now, scope=scope,
                runtime_instance_id=REAL, environment="demo")
            tri[f"{key}={value}"] = [snap.new_entry_allowed,
                                     list(snap.blocking_reasons)]
    emit("[A:tri_state]", tri)

    # --- B. authorization_matches_context ---------------------------------
    ctx = {"environment": "demo", "scope_hash": scope.scope_hash,
           "runtime_instance_id": SENTINEL}
    emit("[B:sentinel_both]",
         authorization_matches_context(
             Decision(expires_at=now + 60.0, environment="demo",
                      scope_hash=scope.scope_hash,
                      runtime_instance_id=SENTINEL),
             ctx, now=now))
    emit("[B:sentinel_context_real_decision]",
         authorization_matches_context(
             Decision(expires_at=now + 60.0, environment="demo",
                      scope_hash=scope.scope_hash,
                      runtime_instance_id=REAL),
             ctx, now=now))

    real_ctx = {"environment": "demo", "scope_hash": scope.scope_hash,
                "runtime_instance_id": REAL}
    real_decision = Decision(expires_at=now + 60.0, environment="demo",
                             scope_hash=scope.scope_hash,
                             runtime_instance_id=REAL)
    emit("[B:real_both]", authorization_matches_context(real_decision, real_ctx,
                                                        now=now))

    # --- B2. 既有比较项逐条未变 -------------------------------------------
    comparisons = {
        "expired": authorization_matches_context(
            Decision(expires_at=now - 1.0, environment="demo",
                     scope_hash=scope.scope_hash, runtime_instance_id=REAL),
            real_ctx, now=now),
        "missing_expiry": authorization_matches_context(
            Decision(expires_at=None, environment="demo",
                     scope_hash=scope.scope_hash, runtime_instance_id=REAL),
            real_ctx, now=now),
        "env_mismatch": authorization_matches_context(
            Decision(expires_at=now + 60.0, environment="live",
                     scope_hash=scope.scope_hash, runtime_instance_id=REAL),
            real_ctx, now=now),
        "scope_mismatch": authorization_matches_context(
            Decision(expires_at=now + 60.0, environment="demo",
                     scope_hash="deadbeef", runtime_instance_id=REAL),
            real_ctx, now=now),
        "runtime_mismatch": authorization_matches_context(
            Decision(expires_at=now + 60.0, environment="demo",
                     scope_hash=scope.scope_hash, runtime_instance_id="other"),
            real_ctx, now=now),
        "empty_context": authorization_matches_context(real_decision, {},
                                                       now=now),
        "none_context": authorization_matches_context(real_decision, None,
                                                      now=now),
    }
    for item in LOOKALIKES:
        comparisons[f"lookalike:{item}"] = authorization_matches_context(
            Decision(expires_at=now + 60.0, environment="demo",
                     scope_hash=scope.scope_hash, runtime_instance_id=item),
            {"environment": "demo", "scope_hash": scope.scope_hash,
             "runtime_instance_id": item}, now=now)
    emit("[B:comparisons]", comparisons)


if __name__ == "__main__":
    main()

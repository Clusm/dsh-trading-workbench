"""U13-FU4 两态可跑探针：锁文件**存在但不可解析**时 InstanceLock 是否 fail-closed。

只使用 ``core.instance_lock.InstanceLock`` 的公开 API（构造 / acquire），因此
在修复前后**同一份探针**都能跑（修复前也会导入成功，只是行为不同）。

用法::

    PYTHONUTF8=1 python probe_lock_corruption.py <tree_root>

``<tree_root>`` 必须含 ``src/core/instance_lock.py``。

跨进程：持有者是**真实子进程**（PID 存活），探针进程扮演"第二个服务"，因此
"两个活持有者"是真实的两进程共存，而不是同进程多对象的近似。

三种情形：
  C1 可解析 + 持有者存活  → 期望拒绝（保持不变）
  C2 可解析 + 持有者已死  → 期望回收（合法遗留锁回收，必须没修坏）
  C3 存在但不可解析       → 期望 fail-closed：拒绝 + **不删除**锁文件

退出码：0 = 全部与期望一致；1 = 存在与期望不一致的用例。
"""
from __future__ import annotations

import json
import os
import subprocess
import sys
import tempfile
from pathlib import Path

# 子进程程序：拿住若干个锁路径并保持存活，直到被杀。
_HOLDER_PROGRAM = r"""
import os, sys, time
sys.path.insert(0, sys.argv[1])
from core.instance_lock import InstanceLock
held = []
for raw in sys.argv[2:]:
    lock = InstanceLock(raw)
    lock.acquire()
    held.append(lock)
print("HELD %d" % os.getpid(), flush=True)
time.sleep(300)
"""


def _child_env() -> dict[str, str]:
    env = os.environ.copy()
    env["PYTHONUTF8"] = "1"
    env["PYTHONIOENCODING"] = "utf-8"
    return env


def _spawn_holder(src_dir: Path, paths: list[Path]) -> tuple[subprocess.Popen[str], int]:
    """启动真实持有者子进程，等它宣告 HELD 后返回 (进程, 真实 PID)。

    Windows 的 venv ``python.exe`` 启动器可能再起一层真实解释器进程，因此持有者
    自报的 ``os.getpid()`` 才是**锁文件里那个 PID**；清理必须按它来杀，否则会留下
    仍持有锁的孤儿进程。
    """
    proc = subprocess.Popen(
        [sys.executable, "-c", _HOLDER_PROGRAM, str(src_dir), *[str(p) for p in paths]],
        stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True,
        encoding="utf-8", errors="replace", env=_child_env(),
    )
    line = (proc.stdout.readline() if proc.stdout else "") or ""
    if "HELD" not in line:
        err = (proc.stderr.read() if proc.stderr else "")[-2000:]
        proc.kill()
        raise RuntimeError(f"持有者子进程未就绪: {line!r} / {err}")
    return proc, int(line.split()[1])


def _kill_holder(proc: subprocess.Popen[str], real_pid: int) -> None:
    """按**真实**解释器 PID 清理持有者（启动器进程可能只是外壳）。"""
    import signal

    for target in (real_pid, proc.pid):
        try:
            os.kill(target, signal.SIGTERM)
        except OSError:
            pass
    try:
        proc.wait(timeout=30)
    except subprocess.TimeoutExpired:
        pass


def _dead_pid() -> int:
    """取一个**确定已死**的真实 PID（真跑一个短命子进程再等它退出）。"""
    proc = subprocess.Popen([sys.executable, "-c", "raise SystemExit(0)"], env=_child_env())
    pid = proc.pid
    proc.wait(timeout=30)
    return pid


def _attempt(path: Path):
    """尝试获取锁；返回 (是否持有, lock 对象, 异常或 None)。"""
    from core.instance_lock import InstanceLock

    lock = InstanceLock(path)
    try:
        lock.acquire()
    except BaseException as exc:  # noqa: BLE001 - 探针要连 KeyboardInterrupt 都记录
        return False, lock, exc
    return True, lock, None


def main() -> int:
    root = Path(sys.argv[1]).resolve()
    src_dir = root / "src"
    sys.path.insert(0, str(src_dir))
    from core.instance_lock import InstanceLock  # noqa: E402

    rows: list[tuple[str, str, str, bool]] = []

    with tempfile.TemporaryDirectory(prefix="fu4-probe-") as tmp:
        base = Path(tmp)
        # 每个子情形一个锁路径，全部由同一个真实持有者子进程持有。
        c1 = base / "c1_intact.lock"
        c3a = base / "c3a_empty.lock"
        c3b = base / "c3b_badjson.lock"
        c3c = base / "c3c_nonutf8.lock"
        c3d = base / "c3d_nopid.lock"
        c3e = base / "c3e_pidzero.lock"
        c2 = base / "c2_stale_dead.lock"

        holder, holder_pid = _spawn_holder(src_dir, [c1, c3a, c3b, c3c, c3d, c3e])
        try:
            # --- 制造"不可解析"的四种形态（持有者仍然存活） -------------------
            c3a.write_bytes(b"")                                   # 空 / 截断为 0 字节
            c3b.write_bytes(b'{"pid": %d,' % holder_pid)            # JSON 坏（截断）
            c3c.write_bytes(b'{"pid": %d}\xff\xfe' % holder_pid)    # 非 UTF-8
            c3d.write_text('{"thread": 5}', encoding="utf-8")       # 无 pid 字段
            c3e.write_text('{"pid": 0}', encoding="utf-8")          # pid 不可用

            for label, path in (("C3a", c3a), ("C3b", c3b), ("C3c", c3c),
                                ("C3d", c3d), ("C3e", c3e)):
                before = path.read_bytes()
                held, lock, exc = _attempt(path)
                alive = InstanceLock._pid_alive(holder_pid)
                if held:
                    observed = (f"接管成功（探针PID={os.getpid()}，"
                                f"持有者PID={holder_pid} 仍存活={alive}）→ 两个活持有者")
                    if lock is not None:
                        lock.release()
                else:
                    observed = f"拒绝 {type(exc).__name__}: {str(exc).splitlines()[0]}"
                untouched = (not path.exists() and before == b"") or (
                    path.exists() and path.read_bytes() == before)
                # 拒绝还必须是**可区分、可行动**的那一种：LockUnreadableError。
                distinguishable = (not held) and type(exc).__name__ == "LockUnreadableError"
                rows.append((label, f"{path.name} 不可解析 → fail-closed",
                             observed, distinguishable and untouched))

            # --- C1：可解析 + 持有者存活 → 拒绝（必须保持不变） ----------------
            c1_before = c1.read_bytes()
            held, lock, exc = _attempt(c1)
            if held:
                observed = f"接管成功（PID={os.getpid()}）→ 两个活持有者"
                lock.release()
            else:
                observed = f"拒绝 {type(exc).__name__}: {str(exc).splitlines()[0]}"
            rows.append(("C1", "可解析+持有者存活 → 拒绝",
                         observed, (not held) and c1.read_bytes() == c1_before))
        finally:
            _kill_holder(holder, holder_pid)

        # --- C2：可解析 + 持有者确定已死 → 允许回收（不能修坏） ---------------
        dead = _dead_pid()
        c2.write_text(json.dumps({"pid": dead, "thread": 1,
                                  "started_at": "2026-01-01T00:00:00+00:00"}),
                      encoding="utf-8")
        assert not InstanceLock._pid_alive(dead), "短命子进程 PID 竟仍存活，探针前提不成立"
        held, lock, exc = _attempt(c2)
        if held:
            payload = json.loads(c2.read_text(encoding="utf-8"))
            observed = (f"回收成功（旧PID={dead} 已死；新PID={payload.get('pid')}）")
            lock.release()
            ok = int(payload.get("pid") or 0) == os.getpid()
        else:
            observed = f"拒绝 {type(exc).__name__}: {str(exc).splitlines()[0]}"
            ok = False
        rows.append(("C2", "可解析+持有者已死 → 回收", observed, ok))

        # --- 正常路径：首进程拿锁 / 次进程被拒 / 释放后可再拿 ------------------
        norm = base / "normal.lock"
        first, _lock, exc_a = _attempt(norm)
        second_ok, second_lock, exc_b = _attempt(norm)
        ok_norm = first and (not second_ok) and isinstance(exc_b, RuntimeError)
        second_lock.release()
        norm.unlink(missing_ok=True)
        third, third_lock, exc_c = _attempt(norm)
        ok_norm = ok_norm and third and exc_c is None
        third_lock.release()
        rows.append(("N", "正常路径：首拿/次拒/释放后再拿",
                     "PASS" if ok_norm else f"FAIL first={first} second={second_ok} "
                     f"third={third}", ok_norm))

    print("\n=== U13-FU4 锁损坏 fail-closed 探针 ===")
    print(f"tree = {root}")
    print(f"{'用例':<5}{'期望':<34}{'观察':<72}结论")
    failed = 0
    for label, expect, observed, ok in rows:
        if not ok:
            failed += 1
        print(f"{label:<5}{expect:<34}{observed:<72}{'PASS' if ok else 'FAIL'}")
    print(f"\nPROBE: {len(rows) - failed} 通过 / {failed} 失败")
    return 1 if failed else 0


if __name__ == "__main__":
    raise SystemExit(main())

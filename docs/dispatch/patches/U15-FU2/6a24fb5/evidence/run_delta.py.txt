"""逐脚本跑统一离线回归清单，落 TSV（rc + 尾部摘要），供两态 delta 对比。"""
from __future__ import annotations

import os
import subprocess
import sys
from pathlib import Path

root = Path(sys.argv[1]).resolve()
out = Path(sys.argv[2]).resolve()
sys.path.insert(0, str(root / "scripts"))
import run_offline_regression as harness  # noqa: E402

env = os.environ.copy()
env["EXECUTION_MODE"] = "paper"
env.pop("MTA_PROFILE", None)
env.pop("MTA_STATE_ROOT", None)
env.pop("MTA_DATA_ROOT", None)
env["MTA_NO_PROXY"] = "1"
env["PYTHONUTF8"] = "1"
env["PYTHONIOENCODING"] = "utf-8"
for name in tuple(env):
    upper = name.upper()
    if ("BINANCE" in upper or "SENSENOVA" in upper) and (
            "KEY" in upper or "SECRET" in upper or "TOKEN" in upper):
        env.pop(name, None)

logs = out.parent / f"{out.stem}_logs"
logs.mkdir(parents=True, exist_ok=True)
rows = []
for index, script in enumerate(harness.SCRIPTS, 1):
    log = logs / (Path(script).stem + ".log")
    with log.open("wb") as handle:
        result = subprocess.run([sys.executable, script], cwd=root, env=env,
                                stdout=handle, stderr=subprocess.STDOUT)
    tail = ""
    try:
        lines = [line for line in log.read_text(encoding="utf-8", errors="replace").splitlines()
                 if line.strip()]
        tail = lines[-1][:160].replace("\t", " ") if lines else ""
    except OSError:
        pass
    rows.append(f"{index}\t{script}\t{result.returncode}\t{tail}")
    print(f"[{index:02d}/{len(harness.SCRIPTS)}] rc={result.returncode} {script}", flush=True)
out.write_text("\n".join(rows) + "\n", encoding="utf-8")
print(f"WROTE {out} rows={len(rows)}")

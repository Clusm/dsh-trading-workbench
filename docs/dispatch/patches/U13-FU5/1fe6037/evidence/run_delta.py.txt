"""U13-FU5 full-delta driver: pristine(base) vs patched(base+diff), 94 scripts.

Per L19 each unpacked tree gets `git init && git add -A` (stage only, no commit),
otherwise smoke_u10a_release_package.py sees an empty `git ls-files` and both
trees collapse identically.
"""
from __future__ import annotations

import importlib.util
import os
import shutil
import subprocess
import sys
import tarfile
from pathlib import Path

LIVE = Path(r"C:/Users/Tao/Desktop/币安/mat-wt/U13F5")
BASE = "1d737b5"
PATCH_DIR = LIVE / "docs/dispatch/patches/U13-FU5/1fe6037"
SBX = Path(r"C:/Users/Tao/AppData/Local/Temp/fu5_delta")
PY = r"C:/Users/Tao/Desktop/币安/multi-agent-trader/.venv/Scripts/python.exe"


def load_scripts() -> tuple[str, ...]:
    spec = importlib.util.spec_from_file_location(
        "ror", LIVE / "scripts/run_offline_regression.py")
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)          # type: ignore[union-attr]
    return tuple(mod.SCRIPTS)


def export(tree: Path, rev: str | None) -> None:
    if tree.exists():
        shutil.rmtree(tree)
    tree.mkdir(parents=True)
    if rev is None:                       # patched = working tree content via tar
        subprocess.run(["git", "-C", str(LIVE), "archive", BASE, "-o",
                        str(SBX / "_base.tar")], check=True)
        with tarfile.open(SBX / "_base.tar") as tf:
            tf.extractall(tree)
        for diff in sorted(PATCH_DIR.glob("0*.diff")):
            subprocess.run(["git", "apply", "--verbose", str(diff)], cwd=tree,
                           check=True, capture_output=True)
        return
    subprocess.run(["git", "-C", str(LIVE), "archive", rev, "-o",
                    str(SBX / "_p.tar")], check=True)
    with tarfile.open(SBX / "_p.tar") as tf:
        tf.extractall(tree)


def stage(tree: Path) -> int:
    subprocess.run(["git", "init", "-q", "."], cwd=tree, check=True)
    subprocess.run(["git", "add", "-A"], cwd=tree, check=True, capture_output=True)
    out = subprocess.run(["git", "ls-files"], cwd=tree, check=True,
                         capture_output=True, text=True, encoding="utf-8")
    return len([line for line in out.stdout.splitlines() if line.strip()])


def env_for() -> dict[str, str]:
    env = os.environ.copy()
    env["EXECUTION_MODE"] = "paper"
    env.pop("MTA_PROFILE", None)
    env.pop("MTA_STATE_ROOT", None)
    env.pop("MTA_DATA_ROOT", None)
    env["MTA_NO_PROXY"] = "1"
    env["PYTHONUTF8"] = "1"
    env["PYTHONIOENCODING"] = "utf-8"
    for name in tuple(env):
        upper = name.upper()
        if ("BINANCE" in upper or "SENSENOVA" in upper) and (
                "KEY" in upper or "SECRET" in upper or "TOKEN" in upper):
            env.pop(name, None)
    return env


def run_tree(tree: Path, scripts: tuple[str, ...], out_tsv: Path,
             log_path: Path) -> dict[str, int]:
    env = env_for()
    results: dict[str, int] = {}
    with log_path.open("w", encoding="utf-8") as log, \
            out_tsv.open("w", encoding="utf-8", newline="\n") as tsv:
        for script in scripts:
            try:
                proc = subprocess.run([PY, script], cwd=tree, env=env,
                                      capture_output=True, text=True,
                                      encoding="utf-8", errors="replace",
                                      timeout=900)
                rc = proc.returncode
                tail = (proc.stdout or "")[-1500:] + (proc.stderr or "")[-1500:]
            except subprocess.TimeoutExpired:
                rc = 124
                tail = "TIMEOUT"
            results[script] = rc
            tsv.write(f"{script}\t{rc}\n")
            log.write(f"=== {script} rc={rc} ===\n{tail}\n\n")
            print(f"  {script} rc={rc}", flush=True)
    return results


def main() -> int:
    scripts = load_scripts()
    SBX.mkdir(parents=True, exist_ok=True)
    pristine, patched = SBX / "pristine", SBX / "patched"
    export(pristine, BASE)
    export(patched, None)
    n_pristine = stage(pristine)
    n_patched = stage(patched)
    print(f"ls_files pristine={n_pristine} patched={n_patched}", flush=True)

    ev = LIVE / "docs/dispatch/patches/U13-FU5/1fe6037/evidence"
    ev.mkdir(parents=True, exist_ok=True)
    r1 = run_tree(pristine, scripts, ev / "reg_pristine.tsv",
                  SBX / "pristine.log")
    r2 = run_tree(patched, scripts, ev / "reg_patched.tsv", SBX / "patched.log")

    new_fail = sorted(s for s in scripts if r1[s] == 0 and r2[s] != 0)
    fixed = sorted(s for s in scripts if r1[s] != 0 and r2[s] == 0)
    both = sorted(s for s in scripts if r1[s] != 0 and r2[s] != 0)
    changed = sorted(s for s in scripts if r1[s] != 0 and r2[s] != 0 and r1[s] != r2[s])
    with (ev / "full_delta_summary.txt").open("w", encoding="utf-8", newline="\n") as fh:
        fh.write(f"pristine={BASE} patched=1fe6037\n")
        fh.write(f"git_ls_files_pristine={n_pristine}\n")
        fh.write(f"git_ls_files_patched={n_patched}\n")
        fh.write(f"pristine_total={len(scripts)} patched_total={len(scripts)}\n")
        fh.write(f"pristine_nonzero={sum(1 for v in r1.values() if v)}\n")
        fh.write(f"patched_nonzero={sum(1 for v in r2.values() if v)}\n")
        fh.write(f"sets_equal={set(r1) == set(r2)}\n")
        fh.write(f"NEW_FAIL={new_fail}\n")
        fh.write(f"FIXED={fixed}\n")
        fh.write(f"BOTH_NONZERO={both}\n")
        fh.write(f"CHANGED={changed}\n")
    print("DONE", flush=True)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

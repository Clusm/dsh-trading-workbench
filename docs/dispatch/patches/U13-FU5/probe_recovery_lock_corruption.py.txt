"""U13-FU5 前后对照探针：**真实存活 writer 子进程 + 锁文件损坏** 时，
``create_epoch_bundle`` / ``check_writer_stopped`` 的行为。

     PYTHONUTF8=1 <venv>/python.exe probe_recovery_lock_corruption.py.txt <repo_root>

在 <repo_root>/src 上导入 core.ops.recovery，全程离线、只写临时目录。
逐行打印关键事实，供两态对照（修复前 / 修复后）；不做任何断言，退出码恒 0。
"""
from __future__ import annotations

import json
import os
import sqlite3
import subprocess
import sys
import tempfile
from pathlib import Path

ROOT = Path(sys.argv[1]).resolve()
sys.path.insert(0, str(ROOT / "src"))

from core.ops.recovery import (  # noqa: E402
    RECOVERY_WRITER_RUNNING,
    check_writer_stopped,
    create_epoch_bundle,
    verify_epoch_bundle,
)
from core.runtime_paths import REQUIRED_STATE_FILES, RuntimePaths  # noqa: E402

_HOLDER = """
import sys, time
sys.path.insert(0, sys.argv[1])
from core.instance_lock import InstanceLock
lock = InstanceLock(sys.argv[2])
lock.acquire()
print("HELD %d" % __import__("os").getpid(), flush=True)
time.sleep(300)
"""


def fixture(base: Path) -> RuntimePaths:
    paths = RuntimePaths.resolve(base, profile=None)
    paths.data_dir.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(paths.db_file)
    conn.execute("CREATE TABLE schema_migrations(version INTEGER PRIMARY KEY)")
    conn.execute("INSERT INTO schema_migrations VALUES(1)")
    conn.commit()
    conn.close()
    for name in REQUIRED_STATE_FILES:
        (paths.data_dir / name).write_text(json.dumps({"fixture": name}), encoding="utf-8")
    return paths


def main() -> int:
    env = os.environ.copy()
    env["PYTHONUTF8"] = "1"
    env["PYTHONIOENCODING"] = "utf-8"
    print(f"REPO={ROOT}")
    with tempfile.TemporaryDirectory(prefix="fu5-probe-") as tmp:
        base = Path(tmp)
        paths = fixture(base / "host")
        bundles = base / "bundles"
        proc = subprocess.Popen(
            [sys.executable, "-c", _HOLDER, str(ROOT / "src"), str(paths.lock_file)],
            stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True,
            encoding="utf-8", errors="replace", env=env)
        line = (proc.stdout.readline() if proc.stdout else "") or ""
        if "HELD" not in line:
            print("HOLDER_FAILED", line, (proc.stderr.read() if proc.stderr else "")[-800:])
            proc.kill()
            return 0
        holder_pid = int(line.split()[1])
        print(f"writer_pid={holder_pid} alive={proc.poll() is None}")
        try:
            from core.instance_lock import InstanceLock
            print(f"lock_readable_and_writer_alive="
                  f"{InstanceLock._pid_alive(holder_pid)}")
            # 锁完好 + 存活 → 既有拒绝路径（两态都成立）
            try:
                check_writer_stopped(paths)
                print("intact_lock_check=NO_RAISE")
            except BaseException as exc:  # noqa: BLE001
                print(f"intact_lock_check={getattr(exc, 'code', type(exc).__name__)}")

            # 真实竞态窗口：O_EXCL 已创建、os.write 还没写 → 0 字节
            paths.lock_file.write_bytes(b"")
            try:
                check_writer_stopped(paths)
                print("corrupt_lock_check=NO_RAISE  <-- 把'读不出'当成'写者已停'")
            except BaseException as exc:  # noqa: BLE001
                print(f"corrupt_lock_check={getattr(exc, 'code', type(exc).__name__)}")

            bundle = None
            err = None
            try:
                bundle = create_epoch_bundle(paths, bundles, environment="demo")
            except BaseException as exc:  # noqa: BLE001
                err = exc
            print(f"bundle_created={bundle is not None}"
                  + (f" path={bundle.name}" if bundle else ""))
            if err is not None:
                print(f"bundle_error={getattr(err, 'code', type(err).__name__)}")
            if bundle is not None:
                manifest = json.loads(
                    (bundle / "manifest.json").read_text(encoding="utf-8"))
                print(f"manifest_writer_stopped={manifest.get('writer_stopped')}")
                try:
                    verified = verify_epoch_bundle(bundle, expected_environment="demo")
                    print(f"verify_epoch_bundle_accepted=True schema={verified['schema']}")
                except BaseException as exc:  # noqa: BLE001
                    print(f"verify_epoch_bundle_accepted=False "
                          f"{getattr(exc, 'code', type(exc).__name__)}")
            print(f"writer_still_alive_at_end={InstanceLock._pid_alive(holder_pid)}")
        finally:
            import signal
            for target in (holder_pid, proc.pid):
                try:
                    os.kill(target, signal.SIGTERM)
                except OSError:
                    pass
            try:
                proc.wait(timeout=30)
            except subprocess.TimeoutExpired:
                proc.kill()
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

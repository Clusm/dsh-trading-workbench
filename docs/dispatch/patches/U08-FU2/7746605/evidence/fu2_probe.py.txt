"""U08-R5-FU2 前后对照探针：审计库 ``db.*`` 抛 sqlite3.OperationalError 时执行器行为。

只观察、不修改被测逻辑。两种状态都能跑（修复前 = 异常穿出；修复后 = 局部化 +
fail-closed 跳过并披露），输出用于补丁包的前后对照证据。

运行（cwd = 仓库根）：
    PYTHONUTF8=1 <venv>/python.exe <this>.py.txt
"""
from __future__ import annotations

import os
import sqlite3
import sys
import tempfile
from datetime import datetime, timedelta, timezone
from pathlib import Path

REPO = Path(os.getcwd()).resolve()
sys.path.insert(0, str(REPO / "src"))

import core.strategy_funnel as fm  # noqa: E402
from agents.strategy_trader import run_strategy_trader  # noqa: E402

try:
    from core.strategy_funnel import ScanFunnel, TicketLedger
except ImportError:  # pragma: no cover - 极老基线上不存在
    ScanFunnel = TicketLedger = None  # type: ignore[assignment]


def ticket(tid: str, symbol: str = "BTCUSDT") -> dict:
    return {
        "ticket_id": tid, "signal_id": "sig-" + tid, "strategy_id": "card_a",
        "symbol": symbol, "side": "LONG", "timeframe": "1h", "signal_price": 100.0,
        "entry_zone_low": 99.0, "entry_zone_high": 101.0,
        "expires_at": (datetime.now(timezone.utc) + timedelta(hours=4)).isoformat(),
        "stop_loss_pct": 0.03, "take_profit_pct": 0.08, "time_stop_hours": 48,
    }


class ReadBoomDB:
    """去重读取（跨进程权威）抛 OperationalError。"""

    def signal_ticket_has_order(self, ticket_id: str) -> bool:
        raise sqlite3.OperationalError("database is locked")

    def log_risk_event(self, **kw: object) -> None:
        pass

    def update_signal_ticket(self, **kw: object) -> None:
        pass


class SelectiveReadBoomDB:
    """只对 t-bad 这一张票的读取抛错：验证"坏票不牵连同轮其余票"。"""

    def signal_ticket_has_order(self, ticket_id: str) -> bool:
        if str(ticket_id) == "t-bad":
            raise sqlite3.OperationalError("database is locked")
        return False

    def log_risk_event(self, **kw: object) -> None:
        pass

    def update_signal_ticket(self, **kw: object) -> None:
        pass


class WriteBoomDB:
    """去重读取正常（返回"没下过单"），纯审计写入抛 OperationalError。"""

    def signal_ticket_has_order(self, ticket_id: str) -> bool:
        return False

    def log_risk_event(self, **kw: object) -> None:
        raise sqlite3.OperationalError("database is locked")

    def update_signal_ticket(self, **kw: object) -> None:
        raise sqlite3.OperationalError("database is locked")


class InterruptDB:
    """BaseException 通道：KeyboardInterrupt 必须继续穿出。"""

    def signal_ticket_has_order(self, ticket_id: str) -> bool:
        raise KeyboardInterrupt("SIGINT")

    def log_risk_event(self, **kw: object) -> None:
        pass

    def update_signal_ticket(self, **kw: object) -> None:
        pass


def observe(label: str, **kwargs: object) -> None:
    funnel = kwargs.get("funnel")
    escaped = None
    decision = None
    try:
        decision = run_strategy_trader(**kwargs)  # type: ignore[arg-type]
    except BaseException as exc:  # noqa: BLE001 - 探针要看清"谁穿出来了"
        escaped = f"{type(exc).__module__}.{type(exc).__name__}: {exc}"
    print(f"  {label}")
    print(f"    escaped  = {escaped}")
    if decision is None:
        print("    decision = <未返回>（整轮作废）")
        print("    intents  = <丢失>")
    else:
        print(f"    decision = {decision.decision}")
        print(f"    intents  = {len(decision.intents)}")
        print(f"    reasoning= {decision.reasoning[:220]}")
    if funnel is not None:
        counts = {k: v for k, v in funnel.detail_counts.items() if v}
        print(f"    funnel   = {counts}")


def main() -> int:
    print("== U08-R5-FU2 前后对照探针 ==")
    print("-- A. 去重读取（:184 db.signal_ticket_has_order）抛 OperationalError --")
    ledger_path = Path(tempfile.mkdtemp(prefix="fu2-probe-")) / "led.json"
    ledger = TicketLedger(ledger_path, clock=lambda: 1_700_000_000_000)
    fa = ScanFunnel(round_id="probe-read")
    observe("票据 t-a（审计库读取不可用）", tickets=[ticket("t-a")],
            prices={"BTCUSDT": 100.0}, funnel=fa, db=ReadBoomDB(), ledger=ledger)
    print(f"    ledger.is_consumed('t-a') = {ledger.is_consumed('t-a')}"
          "   （True = 票据被永久烧掉；False = 仅本轮观望）")

    print("-- A2. 只有 t-bad 的读取不可用：坏票跳过、同轮 t-good 照常开仓 --")
    fa2 = ScanFunnel(round_id="probe-read-partial")
    observe("票据 [t-bad, t-good(ETH)]", tickets=[ticket("t-bad", "BTCUSDT"),
                                                  ticket("t-good", "ETHUSDT")],
            prices={"BTCUSDT": 100.0, "ETHUSDT": 100.0}, funnel=fa2,
            db=SelectiveReadBoomDB())

    print("-- B. 纯审计写入（:227/:237 log_risk_event / update_signal_ticket）抛 OperationalError --")
    # 先造好一笔 intent（ETHUSDT 票），再让第二张票（BTCUSDT 已持仓）触发审计写入失败。
    fb = ScanFunnel(round_id="probe-write")
    observe("票据 [t-good(ETH), t-hold(BTC 已持仓)]",
            tickets=[ticket("t-good", "ETHUSDT"), ticket("t-hold", "BTCUSDT")],
            prices={"BTCUSDT": 100.0, "ETHUSDT": 100.0}, funnel=fb, db=WriteBoomDB(),
            portfolio={"positions": [{"symbol": "BTCUSDT", "source": "ai"}]})

    print("-- C. KeyboardInterrupt 必须继续穿出（不得被 except Exception 吞掉） --")
    fc = ScanFunnel(round_id="probe-ki")
    observe("票据 t-ki", tickets=[ticket("t-ki")], prices={"BTCUSDT": 100.0},
            funnel=fc, db=InterruptDB())

    print("-- 注册表 --")
    print(f"    AUDIT_UNAVAILABLE registered = "
          f"{getattr(fm, 'DETAIL_AUDIT_UNAVAILABLE', '<missing>')}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
